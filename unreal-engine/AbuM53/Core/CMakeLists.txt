cmake_minimum_required(VERSION 3.20)
project(AbuM53Core VERSION 0.0.0.4 LANGUAGES CXX)
add_library(AbuM53Core STATIC Source/Core.cpp Source/Render.cpp Source/Spz.cpp)
target_include_directories(AbuM53Core PUBLIC Include)
target_compile_features(AbuM53Core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(AbuM53Core PRIVATE /W4 /EHsc /permissive-)
else()
  target_compile_options(AbuM53Core PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(AbuM53Core PUBLIC Threads::Threads)
set_source_files_properties(../ThirdParty/Zstd/zstddeclib.c PROPERTIES LANGUAGE CXX)
add_library(AbuM53Codecs STATIC Source/Codecs.cpp ../ThirdParty/Zstd/zstddeclib.c)
target_include_directories(AbuM53Codecs PUBLIC Include PRIVATE ../ThirdParty/Zstd)
target_compile_features(AbuM53Codecs PUBLIC cxx_std_17)
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
  target_link_libraries(AbuM53Codecs PRIVATE ZLIB::ZLIB)
  target_compile_definitions(AbuM53Codecs PRIVATE ABUM53_WITH_ZLIB=1)
else()
  target_compile_definitions(AbuM53Codecs PRIVATE ABUM53_WITH_ZLIB=0)
  message(STATUS "ZLIB unavailable: standalone gzip SPZ v1-v3 disabled; SPZ v4 remains available")
endif()
add_executable(M53Convert ../Tools/M53Convert/main.cpp)
target_link_libraries(M53Convert PRIVATE AbuM53Core AbuM53Codecs)
include(CTest)
if(BUILD_TESTING)
  add_executable(AbuM53CoreTests ../Tests/portable/tests.cpp)
  target_link_libraries(AbuM53CoreTests PRIVATE AbuM53Core AbuM53Codecs)
  target_compile_definitions(AbuM53CoreTests PRIVATE ABUM53_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../Tests/portable/fixtures" ABUM53_TEST_GZIP=$<BOOL:${ZLIB_FOUND}>)
  add_test(NAME AbuM53.Core COMMAND AbuM53CoreTests "${CMAKE_CURRENT_BINARY_DIR}/test-output")
endif()
